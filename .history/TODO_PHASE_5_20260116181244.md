### Step 1: Create Theme Config Module
```python
# src/bkx/ui/theme.py
"""Theme management for bkx."""

from mikomn import load_theme, get_theme_names, get_default_theme

# Default theme
DEFAULT_THEME = "sweet-dracula"

# Cache for loaded theme
_current_theme: str = DEFAULT_THEME
_theme_cache: dict = {}

def get_active_theme() -> dict:
    """Get the currently active theme."""
    global _theme_cache
    if not _theme_cache:
        _theme_cache = load_theme(_current_theme) or {}
    return _theme_cache

def set_theme(name: str) -> bool:
    """Set the active theme by name."""
    global _current_theme, _theme_cache
    theme = load_theme(name)
    if theme:
        _current_theme = name
        _theme_cache = theme
        return True
    return False

def cycle_theme() -> str:
    """Cycle to the next available theme."""
    global _current_theme
    names = get_theme_names()
    if not names:
        return _current_theme
    try:
        idx = names.index(_current_theme)
        _current_theme = names[(idx + 1) % len(names)]
    except ValueError:
        _current_theme = names[0]
    _theme_cache.clear()
    return _current_theme

# Convenience accessors
def primary_color() -> str:
    return get_active_theme().get("colors", {}).get("primary", "#3B82F6")

def success_color() -> str:
    return get_active_theme().get("colors", {}).get("success", "#10B981")

def error_color() -> str:
    return get_active_theme().get("colors", {}).get("error", "#EF4444")

def warning_color() -> str:
    return get_active_theme().get("colors", {}).get("warning", "#F59E0B")
```

### Task 4: Update CLI Render for Theme-Aware Rendering

Update `src/bkx/cli/render.py` to use mikomn's Rich theme integration:

```python
# src/bkx/cli/render.py
from rich.console import Console
from rich.theme import Theme
from mikomn.rich import get_theme

class Render:
    def __init__(self, json: bool = False):
        self.json = json
        self._setup_theme()
    
    def _setup_theme(self):
        theme_data = get_theme("sweet-dracula")
        custom_theme = Theme({
            "info": theme_data.styles.get("info", ""),
            "success": theme_data.styles.get("success", ""),
            "warning": theme_data.styles.get("warning", ""),
            "error": theme_data.styles.get("error", ""),
        })
        self.console = Console(theme=custom_theme)
```
